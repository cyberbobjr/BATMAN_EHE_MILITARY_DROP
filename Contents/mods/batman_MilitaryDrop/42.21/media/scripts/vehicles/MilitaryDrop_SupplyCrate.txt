/* Caisse de largage de Military Drop : un « véhicule » sans roues ni moteur,
   dont le seul intérêt est son coffre (modèle de départ : la remorque vanilla,
   vehicle_trailer.txt).

   Maillage : common/media/models_X/vehicles/Vehicle_MilitaryDrop_SupplyCrate.fbx,

   Y vers le haut, origine au centre du dessous). Avec scale = 0.01 ici et
   scale = 1.0 dans le véhicule, les valeurs ci-dessous sont en mètres :
   1,10 × 0,88 × 1,10 m.

   Texture originale (source/supply_crate/make_texture.py) ; masque noir : pas
   de peinture de carrosserie aléatoire. Salissures et dégâts : textures vanilla.

   Contenu du coffre : distribution vide (MilitaryDrop_Crate.lua) complétée par
   OnFillContainer, qui lit les options sandbox au moment du largage. */
module Base
{
    model Vehicle_MilitaryDrop_SupplyCrate
    {
        mesh = vehicles/Vehicle_MilitaryDrop_SupplyCrate,
        shader = vehicle,
        invertX = false,
        scale = 0.01,
    }

    vehicle MilitaryDrop_SupplyCrate
    {
        neverSpawnKey = true,
        mechanicType = 1,
        offRoadEfficiency = 0.8,
        engineRepairLevel = 4,
        playerDamageProtection = 0,
        model

            file = Vehicle_MilitaryDrop_SupplyCrate,
            scale = 1.0,
            offset = 0.0 0.0 0.0,

        skin
        {
            texture = Vehicles/MilitaryDrop/Vehicle_MilitaryDrop_SupplyCrate,

        /* Zone d'accès au coffre : le centre du personnage doit y entrer

        area TruckBed
        {
            xywh = 0.0 0.0 2.5 2.5,

            container

                capacity = 150,

        }
